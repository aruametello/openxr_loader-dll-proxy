#
# configuration file for controller offsets in OpenXR
#
# One [device:...] section per device, named after its OpenXR
# subaction path. Rotation is written in degrees, position in metres.
#
#   rotation_x = "up/down" angle
#   rotation_y = "left/right" angle
#   rotation_z = "twist" angle
#
# Hands only: smoothing of small movements, position and rotation apart.
#   smooth_position / smooth_rotation = 0 or 1
#   smooth_*_samples = 1 to 5 (the newest counts)
#   smooth_*_strength = 0.0 (raw) to 1.0 (fully smoothed)
#   smooth_position_scale in metres, smooth_rotation_scale in degrees:
#   an older sample this far from the newest weighs half.
#
# Anything you add by hand survives as long as it is a comment.
#
